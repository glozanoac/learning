\section{Context: CUTE / CUTEst}
CUTE (\emph{Constrained and Unconstrained Testing Environment}) is a collection of
optimization test problems and Fortran tools~\cite{BCGT95}.
Its successors are CUTEr~\cite{GOT03} and the current CUTEst~\cite{GOT15}.
Each problem is written in the \emph{Standard Input Format} (SIF) and has a
classification string \texttt{XXXr-XX-n-m}. Hager and Zhang use PALMER1C from
CUTEr as their motivating example~\cite[pp.~2150--2152]{HZ13}.

\section{Origin of \texttt{PALMER1C}}
\begin{itemize}
  \item \textbf{Type:} linear least-squares problem from chemical kinetics.
  \item \textbf{Model:} H--N=N=N, quantum-chemistry level TZVP+MP2. The data are
        energies $y_i$ (kJ/mol) as a function of an angle $x_i$ (radians).
  \item \textbf{Source:} M.~Palmer, Edinburgh, private communication.
        SIF input by N.~Gould (1990)~\cite{SIF}.
  \item \textbf{Classification:} \texttt{QUR2-RN-8-0}:
        quadratic objective (\texttt{Q}), unconstrained (\texttt{U}),
        regular (\texttt{R}), second derivatives available (\texttt{2}),
        real-world application (\texttt{R}), no internal variables (\texttt{N}),
        $n=8$ variables, $m=0$ constraints.
\end{itemize}

\section{Mathematical formulation}
Let $a=(a_0,a_2,a_4,\dots,a_{14})\in\R^8$ be the coefficients of the even
polynomial
\[
  p(x;a)=\sum_{k=0}^{7} a_{2k}\,x^{2k}
        = a_0+a_2x^2+a_4x^4+\cdots+a_{14}x^{14}.
\]
Given $M=35$ data points $(x_i,y_i)$, the problem is
\[
  \boxed{\;
  \min_{a\in\R^8}\; f(a)=\sum_{i=1}^{35}\bigl(p(x_i;a)-y_i\bigr)^2 .\;}
\]
All variables are free.

\paragraph{Notation.} Write $t_i:=x_i^2$. Define $V\in\R^{35\times 8}$ by
$V_{ik}=t_i^{\,k}$ ($k=0,\dots,7$), a Vandermonde matrix in the nodes $t_i$, so
that $p(x_i;a)=(Va)_i$ and $f(a)=\|Va-y\|_2^2$. Let
\[
  A:=V^{\top}V\in\R^{8\times8},\qquad b:=V^{\top}y\in\R^8,\qquad c:=y^{\top}y\in\R .
\]
\section{\texorpdfstring{$f$}{f} as a quadratic}

\begin{lemma}[Expansion of $f$]\label{lem:exp}
For every $a\in\R^8$,
\[
  f(a)=a^{\top}Aa-2\,b^{\top}a+c .
\]
\end{lemma}

\begin{proof}
Expanding the product,
\[
  f(a)=(Va-y)^{\top}(Va-y)=a^{\top}V^{\top}Va-a^{\top}V^{\top}y-y^{\top}Va+y^{\top}y .
\]
The number $a^{\top}V^{\top}y$ is a scalar, so it equals its transpose
$y^{\top}Va$. Hence $f(a)=a^{\top}(V^{\top}V)a-2(V^{\top}y)^{\top}a+y^{\top}y
=a^{\top}Aa-2b^{\top}a+c$.
\end{proof}

Therefore $f=2\varphi+c$ with $\varphi(a):=\tfrac12a^{\top}Aa-b^{\top}a$, the
quadratic to which the paper applies CG~\cite[p.~2152]{HZ13}; $f$ and $\varphi$
have the same minimizers. The properties of $A$ (symmetry, positive
definiteness, spectrum, conditioning) are computed in the companion notebook
\begin{center}\small\texttt{1800-01-GouldOrbanToint\_2003\_PALMER1C\_Quadratic\_Matrix\_Properties.ipynb}\end{center}

\section{Starting point and solution}
\begin{itemize}
  \item \textbf{Starting point:} $a^{(0)}=(1,\dots,1)^\top$, with
        $f(a^{(0)})\approx 3.453\times 10^{8}$.
  \item \textbf{Optimal value:} the SIF file reports $f^*\approx 9.7605048\times10^{-2}$;
        a least-squares solve (QR/SVD) gives $f^*\approx 9.75980\times10^{-2}$.
  \item \textbf{Minimizer} (SVD-based least squares):
\end{itemize}
\[
\begin{array}{llll}
a_0^*\approx 88.29718, & a_2^*\approx -164.38402, & a_4^*\approx 146.05981, & a_6^*\approx -103.42266,\\[2pt]
a_8^*\approx 55.92619, & a_{10}^*\approx -17.83897, & a_{12}^*\approx 3.04227, & a_{14}^*\approx -0.21407.
\end{array}
\]
