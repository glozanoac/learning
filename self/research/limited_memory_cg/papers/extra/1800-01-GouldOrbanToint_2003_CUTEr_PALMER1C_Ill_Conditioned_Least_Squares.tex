\section{Context: CUTE / CUTEst}
CUTE (\emph{Constrained and Unconstrained Testing Environment}) is a collection of
optimization test problems and Fortran tools, by Bongartz, Conn, Gould and Toint (1995).
Its successors are CUTEr and the current CUTEst (Gould, Orban, Toint, 2015).
Each problem is written in the \emph{Standard Input Format} (SIF) and has a
classification string \texttt{XXXr-XX-n-m}.

\section{Origin of \texttt{PALMER1C}}
\begin{itemize}
  \item \textbf{Type:} linear least-squares problem from chemical kinetics.
  \item \textbf{Model:} H--N=N=N, quantum-chemistry level TZVP+MP2. The data are
        energies $y_i$ (kJ/mol) as a function of an angle $x_i$ (radians).
  \item \textbf{Source:} M.~Palmer, Edinburgh, private communication.
        SIF input by N.~Gould (1990).
  \item \textbf{Classification:} \texttt{QUR2-RN-8-0}:
  \begin{itemize}
    \item \texttt{Q}: quadratic objective;
    \item \texttt{U}: unconstrained;
    \item \texttt{R}: regular (smooth);
    \item \texttt{2}: second derivatives available analytically;
    \item \texttt{R}: real-world application;
    \item \texttt{N}: no internal variables;
    \item \texttt{8}: number of variables $n=8$;
    \item \texttt{0}: number of constraints $m=0$.
  \end{itemize}
\end{itemize}

\section{Mathematical formulation}
Let $a=(a_0,a_2,a_4,\dots,a_{14})\in\mathbb{R}^8$ be the coefficients of an even
polynomial of degree 14,
\[
  p(x;a)=\sum_{k=0}^{7} a_{2k}\,x^{2k}
        = a_0+a_2x^2+a_4x^4+\cdots+a_{14}x^{14}.
\]
Given $M=35$ data points $(x_i,y_i)$, the problem is
\[
  \boxed{\;
  \min_{a\in\mathbb{R}^8}\; f(a)=\sum_{i=1}^{35}\bigl(p(x_i;a)-y_i\bigr)^2
  =\sum_{i=1}^{35}\Bigl(\sum_{k=0}^{7}a_{2k}x_i^{2k}-y_i\Bigr)^2 .\;}
\]
All variables are free (no bounds).

\paragraph{Matrix form.}
Define $A\in\mathbb{R}^{35\times 8}$ by $A_{ik}=x_i^{2k}$, $k=0,\dots,7$
(a Vandermonde matrix in $x_i^2$), and $y\in\mathbb{R}^{35}$. Then
\[
  f(a)=\|Aa-y\|_2^2,\qquad
  \nabla f(a)=2A^{\top}(Aa-y),\qquad
  \nabla^2 f(a)=2A^{\top}A .
\]
The Hessian is constant. Since $\operatorname{rank}A=8$, $A^\top A\succ 0$:
$f$ is strictly convex and the unique minimizer solves the normal equations
\[
  A^{\top}A\,a^{*}=A^{\top}y .
\]

\paragraph{Why it is hard.} The columns $x^{0},x^{2},\dots,x^{14}$ with
$|x|\le 1.88$ are nearly linearly dependent. Numerically
$\kappa(A)\approx 1.1\times 10^{6}$, so $\kappa(A^\top A)\approx 1.3\times10^{12}$.
Methods that depend on the Hessian conditioning (steepest descent, CG without
preconditioning) converge very slowly.

\section{Starting point and solution}
\begin{itemize}
  \item \textbf{Starting point:} $a^{(0)}=(1,1,\dots,1)^\top$, with
        $f(a^{(0)})\approx 3.453\times 10^{8}$.
  \item \textbf{Lower bound:} $f\ge 0$ (least squares).
  \item \textbf{Optimal value:} the SIF file reports $f^*\approx 9.7605048\times10^{-2}$.
        Solving the normal equations (QR / SVD) gives $f^*\approx 9.75980\times10^{-2}$.
  \item \textbf{Minimizer} (computed by SVD-based least squares):
\end{itemize}
\[
\begin{array}{llll}
a_0^*\approx 88.29718, & a_2^*\approx -164.38402, & a_4^*\approx 146.05981, & a_6^*\approx -103.42266,\\[2pt]
a_8^*\approx 55.92619, & a_{10}^*\approx -17.83897, & a_{12}^*\approx 3.04227, & a_{14}^*\approx -0.21407.
\end{array}
\]
The large, alternating coefficients are another sign of ill-conditioning.
